package com.custom.tvlauncher
import android.content.Intent
import android.os.Bundle
import android.widget.Button
import androidx.appcompat.app.AppCompatActivity
class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        val btnYouTube = findViewById<Button>(R.id.btnOpenYouTube)
        btnYouTube.setOnClickListener {
            val intent = packageManager.getLaunchIntentForPackage("com.google.android.youtube.tv")
            if (intent != null) { startActivity(intent) }
        }
    }
    override fun onBackPressed() {
        // Disable back button so user is locked into this launcher as home screen
    }
}
